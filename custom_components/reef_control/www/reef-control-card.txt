class ReefControlCard extends HTMLElement{
 setConfig(c){this.c=c||{};this.open=false;this.draw()}
 set hass(h){this.h=h;this.draw()}
 getCardSize(){return 8}
 find(names,domain="sensor"){if(!this.h)return null;for(const n of names){const x=Object.values(this.h.states).find(s=>s.entity_id.startsWith(domain+".")&&s.entity_id.endsWith("_"+n));if(x)return x}return null}
 ok(s){return s&&!["unknown","unavailable","Nicht konfiguriert","Nicht verfügbar"].includes(String(s.state))}
 cls(s){const x=String(s?.attributes?.status||"").toLowerCase();return x.includes("kritisch")?"bad":(x.includes("zu ")||x.includes("warn"))?"warn":(x==="ok"||x==="normal")?"good":""}
 fmt(s){let v=s.state,n=Number(String(v).replace(",","."));if(Number.isFinite(n)){let d=/temperatur|temperature|salinitat|salinity/.test(s.entity_id)?1:/redox/.test(s.entity_id)?0:2;v=n.toLocaleString("de-DE",{minimumFractionDigits:d,maximumFractionDigits:d})}return v+(s.attributes.unit_of_measurement?" "+s.attributes.unit_of_measurement:"")}
 toggle(s){this.h.callService("switch",s.state==="on"?"turn_off":"turn_on",{entity_id:s.entity_id})}
 draw(){if(!this.h||!this.c)return;if(!this.shadowRoot)this.attachShadow({mode:"open"});
  const A=this.find(["aquarium"]),O=this.find(["betriebsstatus","feeding_status"]),G=this.find(["gesamtstatus","overall_status"]),
  T=this.find(["temperatur","temperature_status"]),S=this.find(["salinitat","salinity_status"]),P=this.find(["ph","ph_status"]),
  R=this.find(["redox","redox_status"]),C=this.find(["leitfahigkeit","conductivity"]),W=this.find(["wasserstand","water_level"]),
  V=this.find(["wasserwerte","water_values"]),I=this.find(["icp","reef_icp"]),AL=this.find(["aktive_alarme","active_alarms"]),
  F=this.find(["futterungsmodus","feeding_mode"],"switch"),M=this.find(["wartungsmodus","maintenance_mode"],"switch");
  const live=[["Temperatur",T,"mdi:thermometer"],["Salinität",S,"mdi:waves"]].filter(x=>this.ok(x[1]));
  const more=[["pH",P,"mdi:ph"],["Redox",R,"mdi:flash-outline"],["Leitfähigkeit",C,"mdi:lightning-bolt-outline"],["Wasserstand",W,"mdi:waves-arrow-up"]].filter(x=>this.ok(x[1]));
  const vals=V?.attributes?.values||{}, keys=[["KH","kh"],["Calcium","calcium"],["Magnesium","magnesium"],["Nitrat","nitrate"],["Phosphat","phosphate"]].map(x=>[x[0],vals[x[1]]]).filter(x=>x[1]?.value!=null);
  const hints=[...(AL?.attributes?.alarms||[]).map(x=>typeof x==="string"?x:(x.message||x.reason||"Aktiver Alarm")),...(G?.attributes?.issues||[])];const uh=[...new Set(hints)].slice(0,5);
  const gs=G?.state||"OK",gc=gs==="Kritisch"?"bad":gs==="Warnung"?"warn":"good";
  this.shadowRoot.innerHTML=`<style>
  ha-card{overflow:hidden;border-radius:24px}.hero{padding:21px;background:radial-gradient(circle at 85% 0%,rgba(0,188,212,.17),transparent 45%)}.top{display:flex;justify-content:space-between}.eye,.st,small{color:var(--secondary-text-color)}.eye,.title{font-weight:700}.eye{font-size:11px;letter-spacing:.14em}.title{font-size:25px;margin:5px 0 10px}.mode{display:inline-flex;gap:7px;align-items:center;padding:6px 10px;border-radius:20px;background:rgba(127,127,127,.12);font-size:13px}.dot{width:8px;height:8px;border-radius:50%;background:#66bb6a;box-shadow:0 0 10px #66bb6a}.warn .dot{background:#fbc02d}.bad .dot{background:#ef5350}.right{text-align:right;font-size:12px;color:var(--secondary-text-color)}.right b{display:block;font-size:16px;color:var(--primary-text-color)}
  .sec{padding:0 16px 17px}.h{font-size:12px;font-weight:700;letter-spacing:.05em;color:var(--secondary-text-color);margin:5px 2px 9px;text-transform:uppercase}.grid,.actions,.mini{display:grid;grid-template-columns:1fr 1fr;gap:9px}.live,.box,.iv{background:rgba(127,127,127,.08);border-radius:17px;padding:13px}.live .v{font-size:26px;font-weight:750;margin:8px 0 2px}.good .v{color:#66bb6a;text-shadow:0 0 15px rgba(102,187,106,.45)}.warn .v{color:#fbc02d;text-shadow:0 0 15px rgba(251,192,45,.4)}.bad .v{color:#ef5350;text-shadow:0 0 15px rgba(239,83,80,.45)}.lbl{display:flex;gap:6px;align-items:center;font-size:12px;color:var(--secondary-text-color)}
  button{border:0;color:var(--primary-text-color);font:inherit}.act{padding:13px;border-radius:15px;background:rgba(127,127,127,.1)}.act.on{background:rgba(0,188,212,.18);color:var(--primary-color)}.notes{padding:11px 13px;border-radius:15px;background:rgba(127,127,127,.07);font-size:13px}.note{margin:5px 0}.yes{color:#66bb6a}.icp{display:grid;grid-template-columns:repeat(5,1fr);gap:6px}.iv{text-align:center;padding:9px 4px}.iv span{display:block;font-size:10px;color:var(--secondary-text-color)}.iv b{font-size:15px}.iv small{font-size:9px}.expand{width:100%;padding:13px;background:transparent;border-top:1px solid rgba(127,127,127,.12);color:var(--secondary-text-color)}.details{display:${this.open?"block":"none"};padding:0 16px 17px}.box{display:flex;gap:8px;align-items:center}.box span,.box b{display:block}.box span{font-size:10px;color:var(--secondary-text-color)}@media(max-width:430px){.icp{grid-template-columns:repeat(3,1fr)}}
  </style><ha-card>
  <div class="hero"><div class="top"><div><div class="eye">REEF CONTROL</div><div class="title">${this.c.title||A?.state||"Aquarium"}</div><div class="mode ${gc}"><i class="dot"></i>${O?.state||"Normalbetrieb"}</div></div><div class="right">Gesamtstatus<b>${gs}</b></div></div></div>
  ${live.length?`<div class="sec"><div class="h">Live-Wasserwerte</div><div class="grid">${live.map(x=>`<div class="live ${this.cls(x[1])}"><div class="lbl"><ha-icon icon="${x[2]}"></ha-icon>${x[0]}</div><div class="v">${this.fmt(x[1])}</div><div class="st">${x[1].attributes.status||""}</div></div>`).join("")}</div></div>`:""}
  ${F||M?`<div class="sec"><div class="h">Schnellaktionen</div><div class="actions">${F?`<button id="f" class="act ${F.state==="on"?"on":""}">🐟 Fütterung</button>`:""}${M?`<button id="m" class="act ${M.state==="on"?"on":""}">🔧 Wartung</button>`:""}</div></div>`:""}
  <div class="sec"><div class="h">Aktuelle Hinweise</div><div class="notes">${uh.length?uh.map(x=>`<div class="note">⚠ ${x}</div>`).join(""):`<div class="yes">✓ Keine aktuellen Hinweise</div>`}</div></div>
  ${keys.length?`<div class="sec"><div class="h">Letzte Wasserwerte / ICP</div><div class="icp">${keys.map(x=>`<div class="iv"><span>${x[0]}</span><b>${Number(x[1].value).toLocaleString("de-DE",{maximumFractionDigits:3})}</b><small>${x[1].unit||""}</small></div>`).join("")}</div>${I?`<small>ICP: ${I.state}${I.attributes.provider?" · "+I.attributes.provider:""}${I.attributes.analysis_date?" · "+I.attributes.analysis_date:""}</small>`:""}</div>`:""}
  ${more.length?`<button id="e" class="expand">${this.open?"Weniger anzeigen ▲":"Weitere Werte anzeigen ▼"}</button><div class="details"><div class="mini">${more.map(x=>`<div class="box"><ha-icon icon="${x[2]}"></ha-icon><div><span>${x[0]}</span><b>${this.fmt(x[1])}</b></div></div>`).join("")}</div></div>`:""}
  </ha-card>`;
  this.shadowRoot.getElementById("f")?.addEventListener("click",()=>this.toggle(F));this.shadowRoot.getElementById("m")?.addEventListener("click",()=>this.toggle(M));this.shadowRoot.getElementById("e")?.addEventListener("click",()=>{this.open=!this.open;this.draw()})
 }}
customElements.define("reef-control-card",ReefControlCard);
window.customCards=window.customCards||[];window.customCards.push({type:"reef-control-card",name:"Reef Control Card",description:"Aquarium-Cockpit für Reef Control",preview:true});
console.info("REEF CONTROL CARD v0.1.0");
